\textbf{步骤4}：分解第一�?

\begin{definition}[Sigmoid函数]

\subsubsection{问题设定}

\subsubsection{投影分析}

\begin{align*}

\textbf{展开}�?

\end{align*}

\begin{align*}
\tilde{\sigma}_0^2 + \tilde{\sigma}_1^2 &= \mathbf{w}^T\Sigma_0\mathbf{w} + \mathbf{w}^T\Sigma_1\mathbf{w} \\

\end{align*}

\[

\end{itemize}

\[
