\subsection*{2. IRQ0 中断服务程序 (播放/停止切换)}
\begin{center}
\begin{tikzpicture}[flowchart]
    % Main Column
    \node (start) [startstop] {IRQ0 入口};
    \node (decRecord) [decision, below=of start] {系统录制中? (STAT=3)};
    
    % Right Branch (No recording)
    \node (resetIDX) [process, right=of decRecord, xshift=1.5cm] {重置播放索引};
    \node (decPlay) [decision, below=of resetIDX] {当前状态 (STAT=0)};
    \node (setPlay) [process, right=of decPlay, xshift=0.5cm] {STAT=1, 开始播放};
    \node (setStop) [process, below=of decPlay] {停止播放 (STAT=0)};

    % Determine lowest point of right branch (either setStop or setPlay)
    % setStop is below decPlay. setPlay is right of decPlay. 
    % Visually, setStop is likely lower. We place EXIT below setStop level.
    % We align EXIT horizontally with 'decRecord' (Main Column).
    \path let \p1 = (decRecord), \p2 = (setStop.south) in node (exit) [startstop] at (\x1, \y2 - 2cm) {发送 EOI 并返回};

    % Edges - Main
    \draw [arrow] (start) -- (decRecord);
    
    % Yes -> Exit (Straight down)
    \draw [arrow] (decRecord) -- node[left] {是} (exit);

    % No -> Right Branch
    \draw [arrow] (decRecord) -- node[above] {否} (resetIDX);
    \draw [arrow] (resetIDX) -- (decPlay);
    
    % decPlay logic
    \draw [arrow] (decPlay) -- node[above] {0 (空闲)} (setPlay);
    \draw [arrow] (decPlay) -- node[right] {非0} (setStop);
    
    % Rejoin Exit
    \draw [arrow] (setPlay) |- (exit);
    \draw [arrow] (setStop) |- (exit);

\end{tikzpicture}
\end{center}

\subsection*{5. IRQ5 中断服务程序 (按键录音输入)}
