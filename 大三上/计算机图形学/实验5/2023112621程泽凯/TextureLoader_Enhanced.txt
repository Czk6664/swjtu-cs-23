// 增强版纹理加载函数 - 支持透明PNG等格式
// 替换 ShapeManager.cpp 中的 LoadTextureFromFile 函数

bool LoadTextureFromFile(const wchar_t* filename, ID3D11ShaderResourceView** textureView)
{
    if (!g_pd3dDevice || !filename || !textureView)
        return false;

    *textureView = nullptr;

    // 初始化COM
    CoInitialize(nullptr);

    // 创建WIC工厂
    IWICImagingFactory* pWICFactory = nullptr;
    HRESULT hr = CoCreateInstance(
        CLSID_WICImagingFactory,
        nullptr,
        CLSCTX_INPROC_SERVER,
        IID_PPV_ARGS(&pWICFactory)
    );

    if (FAILED(hr))
        return false;

    // 创建解码器
    IWICBitmapDecoder* pDecoder = nullptr;
    hr = pWICFactory->CreateDecoderFromFilename(
        filename,
        nullptr,
        GENERIC_READ,
        WICDecodeMetadataCacheOnDemand,
        &pDecoder
    );

    if (FAILED(hr))
    {
        pWICFactory->Release();
        return false;
    }

    // 获取第一帧
    IWICBitmapFrameDecode* pFrame = nullptr;
    hr = pDecoder->GetFrame(0, &pFrame);
    if (FAILED(hr))
    {
        pDecoder->Release();
        pWICFactory->Release();
        return false;
    }

    // 获取图片尺寸
    UINT width, height;
    pFrame->GetSize(&width, &height);

    // 创建格式转换器，转换为RGBA格式（保留alpha通道）
    IWICFormatConverter* pConverter = nullptr;
    hr = pWICFactory->CreateFormatConverter(&pConverter);
    if (FAILED(hr))
    {
        pFrame->Release();
        pDecoder->Release();
        pWICFactory->Release();
        return false;
    }

    // 使用RGBA格式以完整保留透明度信息
    hr = pConverter->Initialize(
        pFrame,
        GUID_WICPixelFormat32bppRGBA,
        WICBitmapDitherTypeNone,
        nullptr,
        0.0,
        WICBitmapPaletteTypeCustom
    );

    if (FAILED(hr))
    {
        pConverter->Release();
        pFrame->Release();
        pDecoder->Release();
        pWICFactory->Release();
        return false;
    }

    // 读取像素数据
    UINT stride = width * 4;
    UINT imageSize = stride * height;
    std::vector<BYTE> imageData(imageSize);

    hr = pConverter->CopyPixels(
        nullptr,
        stride,
        imageSize,
        imageData.data()
    );

    // 释放WIC资源
    pConverter->Release();
    pFrame->Release();
    pDecoder->Release();
    pWICFactory->Release();

    if (FAILED(hr))
        return false;

    // 创建D3D11纹理，支持alpha通道
    D3D11_TEXTURE2D_DESC texDesc = {};
    texDesc.Width = width;
    texDesc.Height = height;
    texDesc.MipLevels = 1;
    texDesc.ArraySize = 1;
    texDesc.Format = DXGI_FORMAT_R8G8B8A8_UNORM;  // RGBA格式
    texDesc.SampleDesc.Count = 1;
    texDesc.Usage = D3D11_USAGE_DEFAULT;
    texDesc.BindFlags = D3D11_BIND_SHADER_RESOURCE;

    D3D11_SUBRESOURCE_DATA initData = {};
    initData.pSysMem = imageData.data();
    initData.SysMemPitch = stride;

    ID3D11Texture2D* pTexture = nullptr;
    hr = g_pd3dDevice->CreateTexture2D(&texDesc, &initData, &pTexture);
    if (FAILED(hr))
        return false;

    // 创建着色器资源视图
    D3D11_SHADER_RESOURCE_VIEW_DESC srvDesc = {};
    srvDesc.Format = texDesc.Format;
    srvDesc.ViewDimension = D3D11_SRV_DIMENSION_TEXTURE2D;
    srvDesc.Texture2D.MipLevels = 1;

    hr = g_pd3dDevice->CreateShaderResourceView(pTexture, &srvDesc, textureView);
    pTexture->Release();

    return SUCCEEDED(hr);
}
