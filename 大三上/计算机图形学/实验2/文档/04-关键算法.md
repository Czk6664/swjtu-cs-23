# 关键算法

## 1. 点到线段距离

计算点 P 到线段 AB 的最短距离。

```cpp
float PointToLineDistance(D2D1_POINT_2F point, D2D1_POINT_2F lineStart, D2D1_POINT_2F lineEnd)
{
    float dx = lineEnd.x - lineStart.x;
    float dy = lineEnd.y - lineStart.y;
    float len2 = dx * dx + dy * dy;
    
    if (len2 < 0.0001f) {
        return sqrt((point.x - lineStart.x)² + (point.y - lineStart.y)²);
    }
    
    // 投影参数 t
    float t = ((point.x - lineStart.x) * dx + (point.y - lineStart.y) * dy) / len2;
    t = max(0.0f, min(1.0f, t));
    
    // 投影点
    float nearX = lineStart.x + t * dx;
    float nearY = lineStart.y + t * dy;
    
    return sqrt((point.x - nearX)² + (point.y - nearY)²);
}
```

## 2. 旋转变换

```cpp
D2D1_POINT_2F RotatePoint(D2D1_POINT_2F point, D2D1_POINT_2F center, float angle)
{
    float cos_a = cos(angle);
    float sin_a = sin(angle);
    float dx = point.x - center.x;
    float dy = point.y - center.y;
    
    return D2D1::Point2F(
        center.x + dx * cos_a - dy * sin_a,
        center.y + dx * sin_a + dy * cos_a
    );
}
```

## 3. 缩放变换

```cpp
D2D1_POINT_2F ScalePoint(D2D1_POINT_2F point, D2D1_POINT_2F center, float factor)
{
    return D2D1::Point2F(
        center.x + (point.x - center.x) * factor,
        center.y + (point.y - center.y) * factor
    );
}
```

## 4. 直线-直线交点

```cpp
bool LineLineIntersection(D2D1_POINT_2F p1, D2D1_POINT_2F p2, 
                          D2D1_POINT_2F p3, D2D1_POINT_2F p4, 
                          D2D1_POINT_2F& intersection)
{
    float denom = (p1.x - p2.x) * (p3.y - p4.y) - (p1.y - p2.y) * (p3.x - p4.x);
    if (fabs(denom) < 0.001f) return false;
    
    float t = ((p1.x - p3.x) * (p3.y - p4.y) - (p1.y - p3.y) * (p3.x - p4.x)) / denom;
    float u = -((p1.x - p2.x) * (p1.y - p3.y) - (p1.y - p2.y) * (p1.x - p3.x)) / denom;
    
    if (t < 0.0f || t > 1.0f || u < 0.0f || u > 1.0f) return false;
    
    intersection.x = p1.x + t * (p2.x - p1.x);
    intersection.y = p1.y + t * (p2.y - p1.y);
    return true;
}
```

## 5. 直线-圆交点

```cpp
int LineCircleIntersection(D2D1_POINT_2F p1, D2D1_POINT_2F p2, 
                           D2D1_POINT_2F center, float radius,
                           D2D1_POINT_2F& int1, D2D1_POINT_2F& int2)
{
    float dx = p2.x - p1.x;
    float dy = p2.y - p1.y;
    float fx = p1.x - center.x;
    float fy = p1.y - center.y;
    
    float a = dx * dx + dy * dy;
    float b = 2 * (fx * dx + fy * dy);
    float c = fx * fx + fy * fy - radius * radius;
    
    float discriminant = b * b - 4 * a * c;
    if (discriminant < 0) return 0;
    
    discriminant = sqrt(discriminant);
    float t1 = (-b - discriminant) / (2 * a);
    float t2 = (-b + discriminant) / (2 * a);
    
    int count = 0;
    if (t1 >= 0 && t1 <= 1) {
        int1 = D2D1::Point2F(p1.x + t1 * dx, p1.y + t1 * dy);
        count++;
    }
    if (t2 >= 0 && t2 <= 1 && fabs(t2 - t1) > 0.001f) {
        (count == 0 ? int1 : int2) = D2D1::Point2F(p1.x + t2 * dx, p1.y + t2 * dy);
        count++;
    }
    return count;
}
```

## 6. 圆-圆交点

```cpp
int CircleCircleIntersection(D2D1_POINT_2F c1, float r1, 
                             D2D1_POINT_2F c2, float r2,
                             D2D1_POINT_2F& int1, D2D1_POINT_2F& int2)
{
    float dx = c2.x - c1.x;
    float dy = c2.y - c1.y;
    float dist = sqrt(dx * dx + dy * dy);
    
    if (dist > r1 + r2 || dist < fabs(r1 - r2) || dist < 0.001f) return 0;
    
    float a = (r1 * r1 - r2 * r2 + dist * dist) / (2 * dist);
    float h = sqrt(r1 * r1 - a * a);
    
    float px = c1.x + a * dx / dist;
    float py = c1.y + a * dy / dist;
    
    int1.x = px + h * dy / dist;
    int1.y = py - h * dx / dist;
    int2.x = px - h * dy / dist;
    int2.y = py + h * dx / dist;
    
    return (h > 0.001f) ? 2 : 1;
}
```

## 7. 贝塞尔曲线采样

二次贝塞尔曲线公式：P(t) = (1-t)²P0 + 2(1-t)tP1 + t²P2

```cpp
std::vector<D2D1_POINT_2F> GetSampledPoints(int samplesPerSegment = 10) {
    std::vector<D2D1_POINT_2F> result;
    
    for (size_t i = 0; i < curvePoints.size() - 1; i++) {
        D2D1_POINT_2F p0 = curvePoints[i];
        D2D1_POINT_2F p1 = curvePoints[i + 1];
        D2D1_POINT_2F p2 = (i == curvePoints.size() - 2) ? 
            curvePoints[i + 1] : 
            Midpoint(curvePoints[i + 1], curvePoints[i + 2]);
        
        for (int j = 0; j <= samplesPerSegment; j++) {
            float t = (float)j / samplesPerSegment;
            float u = 1.0f - t;
            result.push_back(D2D1::Point2F(
                u * u * p0.x + 2 * u * t * p1.x + t * t * p2.x,
                u * u * p0.y + 2 * u * t * p1.y + t * t * p2.y
            ));
        }
    }
    return result;
}
```
